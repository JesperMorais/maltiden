```go
// WriteJSON writes a JSON response with the given status code.
func WriteJSON(w http.ResponseWriter, status int, data interface{}) {
    w.Header().Set("Content-Type", "application/json")
    w.WriteHeader(status)
    json.NewEncoder(w).Encode(data)
}

// WriteError writes a JSON error response. Uses json.Marshal to prevent injection
// (fixes VALID-01 — no more string concatenation with err.Error()).
func WriteError(w http.ResponseWriter, status int, code string) {
    w.Header().Set("Content-Type", "application/json")
    w.WriteHeader(status)
    json.NewEncoder(w).Encode(map[string]string{"error": code})
}
```

Then migrate ALL handlers and auth middleware:

**1. Replace all `http.Error(w, '{"error":"..."}', status)` with `WriteError(w, status, "...")`** across every handler file and `pkg/middleware/auth.go`. This is a mechanical replacement — same error codes, same status codes, just using the helper.

**2. Replace all `switch err.Error()` blocks with `errors.Is()` chains** in household.go:
```go
// Before:
switch err.Error() {
case "invalid_code": ...
case "already_member": ...
}
// After:
switch {
case errors.Is(err, domain.ErrInvalidCode):
    WriteError(w, http.StatusBadRequest, "invalid_code")
case errors.Is(err, domain.ErrAlreadyMember):
    WriteError(w, http.StatusConflict, "already_member")
default:
    log.Printf("ERROR [JoinHousehold] %v", err)
    WriteError(w, http.StatusInternalServerError, "internal_error")
}
```
